  reply=
  dec() {
    local x=${1-}
    [[ $x =~ ^[0-9]{1,18}$ ]] || return 1
    reply=$((10#$x))
  }
  positive() {
    dec "${1-}" || return 1
    ((reply > 0))
  }
  parse_time() {
    local x=${1-} n mult=1
    case $x in
      *[sS]) n=${x%?} ;;
      *[mM]) n=${x%?}; mult=60 ;;
      *[hH]) n=${x%?}; mult=3600 ;;
      *[dD]) n=${x%?}; mult=86400 ;;
      *) n=$x ;;
    esac
    dec "$n" || return 1
    ((reply <= 9223372036854775807 / mult)) || return 1
    reply=$((reply * mult))
  }
  parse_size() {
    local x=${1-} n suffix mult=1
    x=${x,,}
    case $x in
      -1|inf|infinite|unlimited) reply=-1; return 0 ;;
    esac
    [[ $x =~ ^([0-9]+)(b|k|kb|m|mb|g|gb)?$ ]] || return 1
    n=${BASH_REMATCH[1]}
    suffix=${BASH_REMATCH[2]}
    case $suffix in
      k|kb) mult=1000 ;;
      m|mb) mult=1000000 ;;
      g|gb) mult=1000000000 ;;
    esac
    dec "$n" || return 1
    ((reply <= 9223372036854775807 / mult)) || return 1
    reply=$((reply * mult))
  }
  parse_pct100() {
    local x=${1%\%} whole frac
    [[ $x =~ ^([0-9]+)([.]([0-9]{1,2}))?$ ]] || return 1
    dec "${BASH_REMATCH[1]}" || return 1
    whole=$reply
    ((whole <= (9223372036854775807 - 99) / 100)) || return 1
    frac=${BASH_REMATCH[3]}
    case ${#frac} in
      0) frac=00 ;;
      1) frac=${frac}0 ;;
    esac
    reply=$((whole * 100 + 10#$frac))
  }
  valid_pkg() {
    [[ ${1-} =~ ^[A-Za-z0-9_]+([.][A-Za-z0-9_]+)+$ ]]
  }
  valid_operand() {
    [[ ${1-} =~ ^[A-Za-z0-9_.:/@+\[\]-]{1,192}$ ]]
  }
  esc() {
    reply=${1//\\/\\\\}
    reply=${reply//$'\t'/\\t}
    reply=${reply//$'\r'/\\r}
    reply=${reply//$'\n'/\\n}
  }
  lenpart() {
    reply="${#1}:$1"
  }
  join4() {
    local a b c d x
    lenpart "$1"; a=$reply
    lenpart "$2"; b=$reply
    lenpart "$3"; c=$reply
    lenpart "$4"; d=$reply
    reply=$a$b$c$d
  }

    date '+%s|%Y%m%d_%H%M%S_%N|%Y%m%d|%Y%m%d%H|%Y-%m-%dT%H:%M:%S%z'
  }
  usage() {
    local row aliases var def parser help shown
    printf '%s\n' \
      'phone_top_profile_v7 [options]' \
      '' \
      'Continuous Android raw top plus bounded read-only component evidence (R12).' \
      'No arguments prints this help and performs no collection.' \
      '' \
      'Options (generated from the option factory):'
    for row in "${OPT[@]}"; do
      IFS='|' read -r aliases var def parser help <<<"$row"
      shown=${aliases//,/ / }
      printf '  %-34s %s' "$shown" "$help"
      case $def in
        empty) printf ' (default empty)' ;;
        @allow0) printf ' (default built-in allowlist)' ;;
        list) ;;
        unset) printf ' (default preset)' ;;
        *) printf ' (default %s)' "$def" ;;
      esac
      printf '\n'
    done
    printf '%s\n' \
      '' \
      'TIME is integer seconds or uses s/m/h/d. SIZE uses B/K/KB/M/MB/G/GB.' \
      'Plain is the default: gzip and base64 are launched only for --encoding compact.' \
      'Every native top is retained by default; exact repeats become timed SEEN records.' \
      'JOBS and PROVIDERS are bounded package/UID projections, not global dumps.' \
      'Runtime staging is private; public TXT files are unique, noclobber, append-only, and daily-rotated.'
  }

  config_error() {
    printf 'CONFIGURATION ERROR: %s\n' "$*" >&2

  normalize_value() {
    local parser=$1 val=$2
    case $parser in
      positive) positive "$val" || return 1 ;;
      time) parse_time "$val" || return 1 ;;
      time_min2)
        parse_time "$val" || return 1
        ((reply >= 2)) || return 1
        ;;
      time_min5)
        parse_time "$val" || return 1
        ((reply >= 5)) || return 1
        ;;
      time_min10)
        parse_time "$val" || return 1
        ((reply >= 10)) || return 1
        ;;
      unsigned) dec "$val" || return 1 ;;
      percent0)
        dec "${val%\%}" || return 1
        ((reply <= 100)) || return 1
        ;;
      percent)
        dec "${val%\%}" || return 1
        ((reply >= 1 && reply <= 100)) || return 1
        ;;
      one_to_32)
        positive "$val" || return 1
        ((reply <= 32)) || return 1
        ;;
      size)
        parse_size "$val" || return 1
        ((reply > 0)) || return 1
        ;;
      size_or_inf)
        case $val in
          unset) reply=unset ;;
          *) parse_size "$val" || return 1 ;;
        esac
        ;;
      size0_inf)
        parse_size "$val" || return 1
        ((reply < 0)) && reply=0
        ;;
      delay)
        case $val in
          native) reply=native ;;
          *) parse_time "$val" || return 1 ;;
        esac
        ;;
      late_threshold)
        if parse_pct100 "$val"; then
          :
        else
          reply=-1
        fi
        ;;
      hour_day_off)
        case $val in hourly|daily|off) reply=$val ;; *) return 1 ;; esac
        ;;
      plain_compact)
        case $val in plain|compact) reply=$val ;; *) return 1 ;; esac
        ;;
      saver)
        case $val in max|high|mid|low|none) reply=$val ;; *) return 1 ;; esac
        ;;
      writable_dir)
        case $val in
          @downloads) reply=${HOME}/storage/downloads ;;
          *) reply=$val ;;
        esac
        [[ -d $reply && -w $reply ]] || return 1
        ;;
      safe_prefix)
        [[ $val =~ ^[A-Za-z0-9_.-]{1,80}$ ]] || return 1
        reply=$val
        ;;
      command_path)
        command -v "$val" >/dev/null 2>&1 || return 1
        reply=$val
        ;;
      serial)
        case $val in
          empty) reply= ;;
          *)
            [[ $val =~ ^[A-Za-z0-9._:-]+$ ]] || return 1
            reply=$val
            ;;
        esac
        ;;
      uid_or_any)
        case $val in
          any) reply=any ;;
          *) dec "$val" || return 1 ;;
        esac
        ;;
      regex)
        ( [[ x =~ $val ]] ) 2>/dev/null
        case $? in 2) return 1 ;; esac
        reply=$val
        ;;
      *) return 1 ;;
    esac
    return 0
  }

              SET[$var]=1
              shift
              ;;
          esac
          break
        fi
      done
      ((found)) && break
    done

      config_error "unknown option: $token"

  bytes_file() {
    reply=$(wc -c <"$1")

  sha_file() {
    local rest
    read -r reply rest < <(sha256sum -- "$1")
    [[ $reply =~ ^[0-9a-f]{64}$ ]]

  host_stat_for_namespace_pid() {
    local wanted=$1 status line self_line= self_depth depth candidate host=
    local -a fields=()
    HOST_PROC_STAT=
    [[ $wanted =~ ^[1-9][0-9]*$ ]] || return 1
    while IFS= read -r line; do
      case $line in NSpid:*) self_line=$line; break ;; esac
    done </proc/self/status
    if [[ -z $self_line ]]; then
      HOST_PROC_STAT=$(<"/proc/$wanted/stat") || return 1
      return 0
    fi
    read -r -a fields <<<"${self_line#NSpid:}"
    self_depth=${#fields[@]}
    ((self_depth > 0)) || return 1
    if ((self_depth == 1)); then
      HOST_PROC_STAT=$(<"/proc/$wanted/stat") || return 1
      return 0
    fi
    for status in /proc/[1-9][0-9]*/status; do
      [[ -r $status ]] || continue
      line=
      while IFS= read -r line; do
        case $line in NSpid:*) break ;; esac
      done <"$status" 2>/dev/null
      case $line in NSpid:*) ;; *) continue ;; esac
      read -r -a fields <<<"${line#NSpid:}"
      depth=${#fields[@]}
      ((depth == self_depth)) || continue
      ((${fields[depth - 1]} == wanted)) || continue
      candidate=${status#/proc/}
      candidate=${candidate%/status}
      case $host in
        ''|"$candidate") host=$candidate ;;
        *) return 1 ;;
      esac
    done
    [[ $host =~ ^[1-9][0-9]*$ ]] || return 1
    HOST_PROC_STAT=$(<"/proc/$host/stat") || return 1
  }
  parse_host_proc_stat() {
    local tail=${HOST_PROC_STAT##*) }
    set -- $tail
    (($# >= 20)) || return 1
    HOST_PROC_STATE=${1-}
    HOST_PROC_START=${20-}
    [[ $HOST_PROC_START =~ ^[0-9]+$ ]]
  }
  pid_start_local() {
    local p=$1
    host_stat_for_namespace_pid "$p" || return 1
    parse_host_proc_stat || return 1
    reply=$HOST_PROC_START
  }
  owned_add() {
    local p=$1 start=$2
    [[ $p =~ ^[1-9][0-9]*$ && $start =~ ^[0-9]+$ ]] || return 1
    OWNED_START[$p]=$start
  }
  owned_remove() {
    unset 'OWNED_START[$1]'

  owned_active() {
    local p=$1 expected=${OWNED_START[$1]-}
    [[ -n $expected ]] || return 1
    pid_start_local "$p" || {
      owned_remove "$p"
      return 1
    }
    [[ $reply == "$expected" ]] || {
      owned_remove "$p"
      return 1
    }
  }

    [[ $p =~ ^[1-9][0-9]*$ ]] || return 0
    host_stat_for_namespace_pid "$p" ||
      { wait "$p" 2>/dev/null; return 0; }
    parse_host_proc_stat || { wait "$p" 2>/dev/null; return 0; }
    [[ $HOST_PROC_START == "$expected" ]] ||
      { wait "$p" 2>/dev/null; return 0; }
    kill "$p" 2>/dev/null
    for ((i=0; i<30; i++)); do
      kill -0 "$p" 2>/dev/null
      case $? in 0) ;; *) break ;; esac
      if host_stat_for_namespace_pid "$p" && parse_host_proc_stat; then
        [[ $HOST_PROC_START == "$expected" ]] || break
        state=$HOST_PROC_STATE
        case $state in Z) break ;; esac
      else
        break
      fi
      sleep 0.1
    done
    kill -0 "$p" 2>/dev/null
    case $? in
      0)
        case $state in
          Z) ;;
          *)
            if host_stat_for_namespace_pid "$p" && parse_host_proc_stat &&
               [[ $HOST_PROC_START == "$expected" ]]; then
              kill -9 "$p" 2>/dev/null
            fi
            ;;
        esac
        ;;
    esac
    wait "$p" 2>/dev/null
  }

  unique_open() {
    local stamp=$1 candidate attempt rc create_fd append_fd
    for ((attempt=0; attempt<64; attempt++)); do
      candidate=$out_dir/${prefix}_${stamp}_${RANDOM}${RANDOM}.txt
      set -o noclobber
      { exec {create_fd}>"$candidate"; } 2>/dev/null
      rc=$?
      set +o noclobber
      if ((rc == 0)); then
        exec {append_fd}>>"/proc/self/fd/$create_fd" 2>/dev/null
        rc=$?
        exec {create_fd}>&-
        if ((rc == 0)); then
          ofd=$append_fd
          file=$candidate
          output_open=1
          return 0
        fi
        rm -f -- "$candidate"
      fi
      ofd=
    done
    return 1
  }

  write_record() {
    local type=$1 body=$2 charge=${3-0} n candidate header footer complete pre rc
    ((output_open)) || return 1
    ((hard_cap_closed == 0 && output_failed == 0)) || return 75
    [[ -f $body && -r $body ]] || return 1
    bytes_file "$body" || return 1
    n=$reply
    candidate=$((record_seq + 1))
    printf -v header '@BEGIN seq=%d type=%s body_bytes=%d\n' "$candidate" "$type" "$n"
    printf -v footer '\n@END seq=%d\n' "$candidate"
    complete=$((${#header} + n + ${#footer}))
    if ((max_bytes > 0 && run_bytes + complete > max_bytes)); then
      hard_cap_closed=1
      set_reason max_bytes
      return 75
    fi
    pre=$file_bytes
    printf '%s' "$header" >&"$ofd" &&
      cat -- "$body" >&"$ofd" &&
      printf '%s' "$footer" >&"$ofd"
    rc=$?
    if ((rc)); then
      truncate -s "$pre" "/proc/self/fd/$ofd" 2>/dev/null
      case $? in
        0) ;;
        *) exec {ofd}>&-; ofd=; output_open=0 ;;
      esac
      output_failed=1
      set_reason write_error
      return 1
    fi
    record_seq=$candidate
    file_bytes=$((file_bytes + complete))
    day_bytes=$((day_bytes + complete))
    hour_bytes=$((hour_bytes + complete))
    run_bytes=$((run_bytes + complete))
    if ((charge)); then
      o_hour_bytes=$((o_hour_bytes + complete))
      o_day_bytes=$((o_day_bytes + complete))
      o_run_bytes=$((o_run_bytes + complete))
    fi
    reply=$candidate
  }

  text_record() {
    local type=$1 meta=$2 charge=${3-0} body=$runtime/text_$BASHPID.$RANDOM
    printf '%s\n' "$meta" >"$body" || return 1
    write_record "$type" "$body" "$charge"
    local rc=$?
    rm -f -- "$body"
    return "$rc"
  }

  saver_open() {
    ((hour_cap < 0 || o_hour_bytes < hour_cap)) &&
      ((day_cap < 0 || o_day_bytes < day_cap)) &&
      ((total_cap < 0 || o_run_bytes < total_cap))
  }
  decay() {
    local now=$1 elapsed
    ((now < duty_at)) && now=$duty_at
    elapsed=$((now - duty_at))
    duty_debt=$((duty_debt - elapsed))
    ((duty_debt < 0)) && duty_debt=0
    duty_at=$now
  }
  observer_open() {
    local now=$((SECONDS * 1000))
    decay "$now"
    ((budget > 0)) || return 1
    saver_open || return 1
    ((duty_debt == 0))
  }
  finish_observer_interval() {
    local start=$1 end=$2 cpu_ms=$3 add
    ((end < start)) && end=$start
    decay "$start"
    if ((budget > 0)); then
      ((cpu_ms <= 92233720368547758)) || cpu_ms=92233720368547758
      add=$((100 * cpu_ms / budget - (end - start)))
      duty_debt=$((duty_debt + add))
      ((duty_debt < 0)) && duty_debt=0
    fi
    duty_at=$end
  }

  clear_prev() {
    local chain p
    for chain in "${PREV_KEYS[@]}"; do
      p=${PREV_PATH[$chain]-}
      case $p in "$runtime"/prev_*.raw) rm -f -- "$p" ;; esac
    done
    SEEN=()
    PREV_STATE=()
    PREV_SIZE=()
    PREV_SHA=()
    PREV_PATH=()
    PREV_KIND=()
    STATE_KIND=()
    PREV_KEYS=()
    admission_closed=0
    seen_count=0
    prev_bytes=0
  }
  advance_prev() {
    local chain=$1 sid=$2 n=$3 sha=$4 kind=$5 stage=$6 retain=$7
    local old=${PREV_PATH[$chain]-} old_n=0 projected new=
    case $old in
      "$runtime"/prev_*.raw) old_n=${PREV_SIZE[$chain]-0} ;;
      *) old= ;;
    esac
    case ${PREV_STATE[$chain]+set} in
      set) ;;
      *) PREV_KEYS+=("$chain") ;;
    esac
    if ((retain == 0)); then
      PREV_STATE[$chain]=$sid
      PREV_SIZE[$chain]=$n
      PREV_SHA[$chain]=$sha
      PREV_KIND[$chain]=$kind
      PREV_PATH[$chain]=
      case $old in
        '') ;;
        *) rm -f -- "$old"; prev_bytes=$((prev_bytes - old_n)) ;;
      esac
      rm -f -- "$stage"
      return 0
    fi
    projected=$((prev_bytes - old_n + n))
    if ((projected <= prev_cap)); then
      ((prev_serial++))
      new=$runtime/prev_$prev_serial.raw
      if mv -- "$stage" "$new"; then
        PREV_STATE[$chain]=$sid
        PREV_SIZE[$chain]=$n
        PREV_SHA[$chain]=$sha
        PREV_KIND[$chain]=$kind
        PREV_PATH[$chain]=$new
        case $old in
          '') ;;
          "$new") ;;
          *) rm -f -- "$old" ;;
        esac
        prev_bytes=$projected
        return 0
      fi
    fi
    case $old in '') ;; *) rm -f -- "$old"; prev_bytes=$((prev_bytes - old_n)) ;; esac
    rm -f -- "$stage"
    unset 'PREV_STATE[$chain]' 'PREV_SIZE[$chain]' 'PREV_SHA[$chain]' \
      'PREV_KIND[$chain]' 'PREV_PATH[$chain]'
    return 0
  }

  common_prefix() {
    local old=$1 fresh=$2 limit=$3 lo=0 hi=$limit mid
    while ((lo < hi)); do
      mid=$(((lo + hi + 1) / 2))
      cmp -s -n "$mid" -- "$old" "$fresh"
      case $? in 0) lo=$mid ;; *) hi=$((mid - 1)) ;; esac
    done
    reply=$lo
  }
  common_suffix() {
    local old=$1 fresh=$2 limit=$3 tag=$4 lo=0 hi=$limit mid
    local a=$runtime/suffix_a_$tag b=$runtime/suffix_b_$tag
    while ((lo < hi)); do
      mid=$(((lo + hi + 1) / 2))
      tail -c "$mid" -- "$old" >"$a" &&
        tail -c "$mid" -- "$fresh" >"$b" &&
        cmp -s -- "$a" "$b"
      case $? in 0) lo=$mid ;; *) hi=$((mid - 1)) ;; esac
    done
    rm -f -- "$a" "$b"
    reply=$lo
  }
  make_delta() {
    local old=$1 fresh=$2 old_n=$3 new_n=$4 tag=$5 min prefix suffix room
    local matched maximum middle rebuilt
    min=$old_n
    ((new_n < min)) && min=$new_n
    common_prefix "$old" "$fresh" "$min"
    prefix=$reply
    room=$((min - prefix))
    common_suffix "$old" "$fresh" "$room" "$tag"
    suffix=$reply
    matched=$((prefix + suffix))
    maximum=$old_n
    ((new_n > maximum)) && maximum=$new_n
    if ((maximum)); then DELTA_SIM=$((100 * matched / maximum)); else DELTA_SIM=100; fi
    ((DELTA_SIM >= delta_match)) || return 1
    DELTA_MIDDLE_BYTES=$((new_n - prefix - suffix))
    ((DELTA_MIDDLE_BYTES < new_n)) || return 1
    DELTA_PREFIX=$prefix
    DELTA_SUFFIX=$suffix
    DELTA_MIDDLE=$runtime/delta_middle_$tag
    DELTA_REBUILT=$runtime/delta_rebuilt_$tag
    dd if="$fresh" of="$DELTA_MIDDLE" bs=1 skip="$prefix" \
      count="$DELTA_MIDDLE_BYTES" status=none || return 1
    {
      ((prefix == 0)) || head -c "$prefix" -- "$old"
      cat -- "$DELTA_MIDDLE"
      ((suffix == 0)) || tail -c "$suffix" -- "$old"
    } >"$DELTA_REBUILT" || return 1
    bytes_file "$DELTA_REBUILT" || return 1
    ((reply == new_n)) || return 1
    cmp -s -- "$fresh" "$DELTA_REBUILT"
  }
  wire_payload() {
    local src=$1 tag=$2 raw_n gz b64
    bytes_file "$src" || return 1
    raw_n=$reply
    WIRE_PATH=$src
    WIRE_MODE=plain
    WIRE_BYTES=$raw_n
    case $encoding in
      compact)
        gz=$runtime/wire_$tag.gz
        b64=$runtime/wire_$tag.b64
        gzip -1 -n -c -- "$src" >"$gz"
        case $? in
          0)
            base64 -w 0 -- "$gz" >"$b64"
            case $? in
              0)
                bytes_file "$b64" || return 1
                if ((reply < raw_n)); then
                  WIRE_PATH=$b64
                  WIRE_MODE=gzip64
                  WIRE_BYTES=$reply
                fi
                ;;
            esac
            ;;
        esac
        rm -f -- "$gz"
        ;;
    esac
  }
  state_header() {
    local sid=$1 domain=$2 chain=$3 kind=$4 raw_n=$5 sha=$6
    local wire_mode=$7 wire_n=$8 observation=${9-} base=${10-}
    local prefix=${11-0} suffix=${12-0} middle=${13-0} sim=${14-0}
    local ec eo extra=
    esc "$chain"; ec=$reply
    esc "$observation"; eo=$reply
    case $kind in
      delta)
        printf -v extra '\tbase=%s\tprefix_bytes=%s\tsuffix_bytes=%s\tmiddle_bytes=%s\tsimilarity=%s' \
          "$base" "$prefix" "$suffix" "$middle" "$sim"
        ;;
    esac
    case $observation in '') ;; *) extra+=$'\tobservation='"$eo" ;; esac
    printf -v STATE_HEADER \
      'id=%s\tdomain=%s\tchain=%s\tkind=%s\traw_bytes=%s\traw_sha256=%s%s\nPAYLOAD wire=%s bytes=%s\n' \
      "$sid" "$domain" "$ec" "$kind" "$raw_n" "$sha" "$extra" \
      "$wire_mode" "$wire_n"
  }
  build_state_body() {
    local body=$1 wire=$8
    state_header "$2" "$3" "$4" "$5" "$6" "$7" "$9" "${10}" \
      "${11-}" "${12-}" "${13-0}" "${14-0}" "${15-0}" "${16-0}"
    {
      printf '%s' "$STATE_HEADER"
      cat -- "$wire"
    } >"$body"
  }

  COMMIT_ID=
  COMMIT_NEW=0
  commit_state() {
    local domain=$1 chain=$2 stage=$3 observation=$4 heavy=$5 charge=$6
    local retain=$7 n sha lookup sid immediate=0 tag full_body delta_body
    local full_wire full_mode full_n old old_n chosen kind=full
    local delta_n logical_full logical_delta
    [[ -f $stage ]] || return 1
    bytes_file "$stage" || return 1
    n=$reply
    sha_file "$stage" || return 1
    sha=$reply
    lookup=$chain$'\x1f'$n$'\x1f'$sha
    sid=${SEEN[$lookup]-}
    if [[ ${PREV_SIZE[$chain]-x} == "$n" && ${PREV_SHA[$chain]-x} == "$sha" ]]; then
      immediate=1
      case $sid in '') sid=${PREV_STATE[$chain]-} ;; esac
    fi
    if [[ -n $sid ]]; then
      if ((immediate)); then
        rm -f -- "$stage"
      else
        advance_prev "$chain" "$sid" "$n" "$sha" "${STATE_KIND[$sid]-full}" \
          "$stage" "$retain"
      fi
      COMMIT_ID=$sid
      COMMIT_NEW=0
      return 0
    fi

    sid=S$((state_serial + 1))
    tag=${sid}_$RANDOM
    wire_payload "$stage" "${tag}_full" || return 1
    full_wire=$WIRE_PATH
    full_mode=$WIRE_MODE
    full_n=$WIRE_BYTES
    full_body=$runtime/state_full_$tag
    build_state_body "$full_body" "$sid" "$domain" "$chain" full "$n" "$sha" \
      "$full_wire" "$full_mode" "$full_n" "$observation" || return 1
    chosen=$full_body

    old=${PREV_PATH[$chain]-}
    old_n=${PREV_SIZE[$chain]-0}
    if ((heavy)) && [[ -f $old ]] && [[ ${PREV_KIND[$chain]-full} == full ]]; then
      make_delta "$old" "$stage" "$old_n" "$n" "$tag"
      case $? in
        0)
          state_header "$sid" "$domain" "$chain" full "$n" "$sha" plain "$n" \
            "$observation"
          logical_full=$((${#STATE_HEADER} + n))
          state_header "$sid" "$domain" "$chain" delta "$n" "$sha" plain \
            "$DELTA_MIDDLE_BYTES" "$observation" "${PREV_STATE[$chain]}" \
            "$DELTA_PREFIX" "$DELTA_SUFFIX" "$DELTA_MIDDLE_BYTES" "$DELTA_SIM"
          logical_delta=$((${#STATE_HEADER} + DELTA_MIDDLE_BYTES))
          if ((logical_delta < logical_full)); then
            wire_payload "$DELTA_MIDDLE" "${tag}_delta" || return 1
            delta_body=$runtime/state_delta_$tag
            build_state_body "$delta_body" "$sid" "$domain" "$chain" delta "$n" "$sha" \
              "$WIRE_PATH" "$WIRE_MODE" "$WIRE_BYTES" "$observation" \
              "${PREV_STATE[$chain]}" "$DELTA_PREFIX" "$DELTA_SUFFIX" \
              "$DELTA_MIDDLE_BYTES" "$DELTA_SIM" || return 1
            bytes_file "$delta_body" || return 1
            delta_n=$reply
            bytes_file "$full_body" || return 1
            if ((delta_n < reply)); then
              chosen=$delta_body
              kind=delta
            fi
          fi
          ;;
      esac
    fi

    write_record STATE "$chosen" "$charge" || return
    state_serial=$((state_serial + 1))
    STATE_KIND[$sid]=$kind
    if ((admission_closed == 0)); then
      SEEN[$lookup]=$sid
      ((seen_count++))
      ((seen_count >= seen_cap)) && admission_closed=1
    fi
    advance_prev "$chain" "$sid" "$n" "$sha" "$kind" "$stage" "$retain"
    rm -f -- "$full_body" "${delta_body-}" "${DELTA_MIDDLE-}" \
      "${DELTA_REBUILT-}" "$runtime/wire_${tag}_full.b64" \
      "$runtime/wire_${tag}_delta.b64"
    COMMIT_ID=$sid
    COMMIT_NEW=1
  }

  occurrence_tuple() {
    local iso=$1 source=$2 partial=$3 raw_only=$4 truncated=$5
    local drop_b=$6 drop_l=$7 x="$iso|source=$source"
    ((partial)) && x+="|partial=1"
    ((raw_only)) && x+="|raw_only=1"
    ((truncated)) && x+="|truncated=1"
    ((drop_b)) && x+="|drop_bytes=$drop_b"
    ((drop_l)) && x+="|drop_lines=$drop_l"
    esc "$x"
  }
  init_occurrence() {
    local sid=$1
    case ${OCC_TOTAL[$sid]+set} in
      set) ;;
      *)
        OCC_TOTAL[$sid]=1
        OCC_PENDING[$sid]=
        OCC_ADDED[$sid]=0
        OCC_ORDER+=("$sid")
        ;;
    esac
  }
  flush_occurrences() {
    local sid added body tuple rc
    ((max_bytes == 0)) || return 0
    for sid in "${OCC_ORDER[@]}"; do
      added=${OCC_ADDED[$sid]-0}
      ((added)) || continue
      body=$runtime/seen_$RANDOM
      tuple=${OCC_PENDING[$sid]}
      printf 'state=%s\ttotal=%s\tadded=%s\tat={%s}\n' \
        "$sid" "${OCC_TOTAL[$sid]}" "$added" "$tuple" >"$body" || return 1
      write_record SEEN "$body" 0 || {
        rc=$?
        rm -f -- "$body"
        return "$rc"
      }
      saved_n=$((saved_n + added))
      OCC_PENDING[$sid]=
      OCC_ADDED[$sid]=0
      rm -f -- "$body"
    done
  }
  add_top_occurrence() {
    local sid=$1 tuple=$2 body rc
    init_occurrence "$sid"
    if ((max_bytes > 0)); then
      body=$runtime/seen_one_$RANDOM
      printf 'state=%s\ttotal=%s\tadded=1\tat={%s}\n' \
        "$sid" "$((OCC_TOTAL[$sid] + 1))" "$tuple" >"$body" || return 1
      write_record SEEN "$body" 0 || {
        rc=$?
        rm -f -- "$body"
        return "$rc"
      }
      OCC_TOTAL[$sid]=$((OCC_TOTAL[$sid] + 1))
      saved_n=$((saved_n + 1))
      rm -f -- "$body"
    else
      OCC_TOTAL[$sid]=$((OCC_TOTAL[$sid] + 1))
      OCC_ADDED[$sid]=$((OCC_ADDED[$sid] + 1))
      OCC_PENDING[$sid]+="${OCC_PENDING[$sid]:+;}$tuple"
    fi
  }

  ro() {

  read_device_identity() {
    local state actual boot n clk
    state=$(timeout -k 2s "${diag_to}s" "${A[@]}" get-state 2>/dev/null)
    [[ $state == device ]] || return 2
    actual=$(ro "$diag_to" id -u 2>/dev/null)
    actual=${actual//$'\r'/}
    actual=${actual//$'\n'/}
    [[ $actual =~ ^[0-9]+$ ]] || return 2
    case $shell_uid in
      any) ;;
      *) [[ $actual == "$shell_uid" ]] || return 2 ;;
    esac
    boot=$(ro "$diag_to" cat /proc/sys/kernel/random/boot_id 2>/dev/null)
    boot=${boot//$'\r'/}
    boot=${boot//$'\n'/}
    [[ $boot =~ ^[A-Fa-f0-9]{8}-[A-Fa-f0-9]{4}-[A-Fa-f0-9]{4}-[A-Fa-f0-9]{4}-[A-Fa-f0-9]{12}$ ]] ||
      return 2
    n=$(ro "$diag_to" nproc 2>/dev/null)
    n=${n//$'\r'/}
    if dec "$n" && ((reply > 0 && reply < 1024)); then
      RECHECK_NPROC=$reply
    else
      RECHECK_NPROC=1
    fi
    clk=$(ro "$diag_to" getconf CLK_TCK 2>/dev/null)
    clk=${clk//$'\r'/}
    if dec "$clk" && ((reply > 0)); then
      RECHECK_CLK=$reply
    else
      RECHECK_CLK=100
    fi
    RECHECK_UID=$actual
    RECHECK_BOOT=$boot
  }
  connect_device() {
    local -a devices=()
    if [[ -z $serial ]]; then
      mapfile -t devices < <(
        timeout -k 2s "${diag_to}s" "$adb" devices 2>/dev/null |
          awk 'NR>1 && $2=="device" {print $1}'
      )
      ((${#devices[@]} == 1)) || {
        config_error "use --serial; ready adb devices=${#devices[@]}"
        return 2
      }
      serial=${devices[0]}
    fi
    A=("$adb" -s "$serial")
    read_device_identity || {
      config_error 'adb target identity or boot ID is unavailable'
      return 2
    }
    adb_uid=$RECHECK_UID
    boot_id=$RECHECK_BOOT
    device_nproc=$RECHECK_NPROC
    remote_clk=$RECHECK_CLK
    local_clk=$(getconf CLK_TCK 2>/dev/null)
    if dec "$local_clk" && ((reply > 0)); then
      local_clk=$reply
    else
      local_clk=100
    fi
  }

  resolve_entity() {
    local linux_user=$1 args=$2 pid=$3 observed_uid=$4

    rc=125
    if [[ $child =~ ^[1-9][0-9]*$ ]]; then
      pid_start_local "$child"

      worker_capture_component "$pre" 65536 identity "$pid" "$user" "$pkg" - \
        0 "$uid" "$cmd"
      read -r _ start got_uid < <(awk '/^V7_IDENTITY [0-9]+ [0-9]+$/ {print; exit}' "$pre")
      if [[ $start =~ ^[0-9]+$ && $got_uid == "$uid" ]]; then
        target_quality=complete
      else
        target_quality=terminal
      fi
      ci=0
      : >"$worker_stage/target_$i.tsv"
      if [[ $target_quality == complete ]]; then
        for row in "${CAPTURE[@]}"; do
          IFS='|' read -r scope need comp cap gate recipe operand <<<"$row"
          case $scope in proc|pkg|deep) ;; *) continue ;; esac
          ((depth >= need)) || continue
          case $gate in
            packages) ((packages)) || continue ;;
          esac
          case $scope:$pkg in pkg:-) continue ;; esac
          ((ci++))
          worker_capture_component "$worker_stage/t_${i}_${ci}.raw" "$cap" \
            "$recipe" "$pid" "$user" "$pkg" "$operand" "$start" "$uid" "$cmd"
          read -r rc quality <"$worker_stage/t_${i}_${ci}.raw.meta"
          printf '%s\t%s\t%s\t%s\n' "$ci" "$comp" "$rc" "$quality" \
            >>"$worker_stage/target_$i.tsv"
          case $quality in incomplete) target_quality=incomplete ;; esac
        done
        post=$worker_stage/post_$i.raw
        worker_capture_component "$post" 65536 identity "$pid" "$user" "$pkg" - \
          "$start" "$uid" "$cmd"
        if grep -q "^V7_IDENTITY $start $uid$" "$post"; then
          :
        else
          target_quality=terminal
        fi
      fi
      printf '%s\t%s\n' "$target_quality" "${start:-0}" \
        >"$worker_stage/target_$i.status"
    done
    if ((worker_include_context)); then
      ci=0
      : >"$worker_stage/context.tsv"
      printf '%s\n' complete >"$worker_stage/context.status"
      for row in "${CAPTURE[@]}"; do
        IFS='|' read -r scope need comp cap gate recipe operand <<<"$row"
        [[ $scope == ctx ]] || continue
        case $gate in
          context) ((context)) || continue ;;
          logcat) ((with_logcat)) || continue ;;
        esac
        ((ci++))
        worker_capture_component "$worker_stage/c_${ci}.raw" "$cap" \
          "$recipe" 0 0 - "$operand" 0 - -
        read -r rc quality <"$worker_stage/c_${ci}.raw.meta"
        printf '%s\t%s\t%s\t%s\n' "$ci" "$comp" "$rc" "$quality" \
          >>"$worker_stage/context.tsv"
        case $quality in
          complete) ;;
          incomplete) printf '%s\n' incomplete >"$worker_stage/context.status" ;;
          *)
            case $(<"$worker_stage/context.status") in
              complete) printf '%s\n' "$quality" >"$worker_stage/context.status" ;;
            esac
            ;;
        esac
      done
    fi
  }
  queue_pop() {
    local class=$1 key now=$(date +%s)
    reply=
    case $class in
      M)
        while ((qm_head <= qm_tail)); do
          key=${QM[$qm_head]-}
          unset 'QM[$qm_head]'
          ((qm_head++))
          [[ ${JOB_PHASE[$key]-} == queued ]] || continue
          if (( ${JOB_READY[$key]-0} > now )); then
            ((qm_tail++))
            QM[$qm_tail]=$key
            continue
          fi
          reply=$key
          return 0
        done
        ;;
      O)
        while ((qo_head <= qo_tail)); do
          key=${QO[$qo_head]-}
          unset 'QO[$qo_head]'
          ((qo_head++))
          [[ ${JOB_PHASE[$key]-} == queued ]] || continue
          reply=$key
          return 0
        done
        ;;
    esac
    return 1
  }
  prepare_retries() {
    local key now=$(date +%s)
    for key in "${JOB_KEYS[@]}"; do
      [[ ${JOB_PHASE[$key]-} == retry_wait ]] || continue
      (( ${JOB_READY[$key]-0} <= now )) || continue
      JOB_PHASE[$key]=queued
      append_retry "$key"
    done
  }
  active_fresh() {
    local now=$(date +%s)
    ((active_at > 0 && now >= active_at && now - active_at <= context_ttl))
  }
  entity_active() {
    local entity=$1 pid=${2-${ENTITY_PID[$1]-0}} uid=${3-${ENTITY_UID[$1]--}}
    local user=${4-${ENTITY_USER[$1]--}} pkg=${5-${ENTITY_PKG[$1]--}}
    local label=${ENTITY_LABEL[$entity]-} pair
    [[ $label =~ $allow ]] && return 0
    case $pid in
      ''|0|-) ;;
      *) [[ ${ACTIVE[pid:$pid]-0} == 1 ]] && return 0 ;;
    esac
    case $uid in
      ''|-) ;;
      *) [[ ${ACTIVE[uid:$uid]-0} == 1 ]] && return 0 ;;
    esac
    case $user:$pkg in
      *:-|*:) ;;
      *)
        pair=$user$'\x1f'$pkg
        [[ ${ACTIVE[upkg:$pair]-0} == 1 ]] && return 0
        ;;
    esac
    return 1
  }
  o_target_gate() {
    local key=$1 entity=${JOB_ENTITY[$1]-} now=$((SECONDS * 1000))
    decay "$now"
    ((watch && context && budget > 0)) || return 1
    saver_open || return 1
    active_fresh || return 2
    entity_active "$entity" "${JOB_PID[$key]-0}" "${JOB_UID[$key]--}" \
      "${JOB_USER[$key]--}" "${JOB_PKG[$key]--}" && return 1
    mandatory_exists && return 3
    ((duty_debt == 0)) || return 3
    return 0
  }
  resolve_o_candidates() {
    local quality=$1 key gate
    o_refresh_due=0
    for key in "${JOB_KEYS[@]}"; do
      case ${JOB_TIER[$key]-}:${JOB_PHASE[$key]-} in
        O:context_wait) ;;
        *) continue ;;
      esac
      case $quality in
        complete)
          o_target_gate "$key"
          gate=$?
          case $gate in
            0|3)
              JOB_PHASE[$key]=queued
              append_retry "$key"
              ;;
            2) o_refresh_due=1 ;;
            *) job_unset "$key" ;;
          esac
          ;;
        *)
          job_unset "$key"
          ;;
      esac
    done
  }
  launch_worker() {
    local class= key entity i=0 gate context_mode=none
    [[ -z $worker_pid ]] || return 0
    prepare_retries
    if mandatory_exists; then
      class=M
    elif ((context_due)); then
      class=M
      context_mode=mandatory
    elif ((o_refresh_due)); then
      waiting_optional_exists || { o_refresh_due=0; return 0; }
      ((watch && context && budget > 0)) || {
        resolve_o_candidates canceled
        return 0
      }
      observer_open || {
        saver_open || resolve_o_candidates canceled
        return 0
      }
      class=O
      context_mode=optional
    else
      class=O
      observer_open || return 0
    fi
    ((batch_serial++))
    worker_stage=$runtime/worker_$batch_serial
    mkdir -p -- "$worker_stage" || return 1
    worker_class=$class
    worker_context_mode=$context_mode
    worker_include_context=0
    case $context_mode in mandatory|optional) worker_include_context=1 ;; esac
    if [[ $context_mode == none ]]; then
      while ((i < batch_cap)); do
        queue_pop "$class" || break
        key=$reply
        if [[ $class == O ]]; then
          o_target_gate "$key"
          gate=$?
          case $gate in
            0) ;;
            1) job_unset "$key"; continue ;;
            2)
              JOB_PHASE[$key]=context_wait
              o_refresh_due=1
              continue
              ;;
            *)
              append_retry "$key"
              break
              ;;
          esac
        fi
        entity=${JOB_ENTITY[$key]}
        ((i++))
        B_JOB[$i]=$key
        B_PID[$i]=${JOB_PID[$key]-0}
        case ${JOB_TIER[$key]} in H|O) B_DEPTH[$i]=2 ;; *) B_DEPTH[$i]=1 ;; esac
        B_USER[$i]=${JOB_USER[$key]--}
        B_PKG[$i]=${JOB_PKG[$key]--}
        B_UID[$i]=${JOB_UID[$key]--}
        B_CMD[$i]=${JOB_CMD[$key]--}
        JOB_PHASE[$key]=flight
      done
    fi
    worker_count=$i
    if ((worker_count == 0 && worker_include_context == 0)); then
      rmdir -- "$worker_stage" 2>/dev/null
      worker_stage=
      return 0
    fi
    worker_started_ms=$((SECONDS * 1000))
    worker_child_note=$worker_stage/ACTIVE_CHILD
    cpu_ticks_self
    worker_ctrl_start=$reply
    worker_source_cpu_start=0
    worker_source_cpu_tuple=
    if [[ $source_pid =~ ^[1-9][0-9]*$ ]]; then
      pid_start_local "$source_pid"
      if [[ $reply == "$source_start" ]]; then
        pid_cpu_ticks "$source_pid"
        worker_source_cpu_start=$reply
        worker_source_cpu_tuple=$source_pid:$source_start
      fi
    fi
    coproc V7_WORKER {
      worker_main
      rc=$?
      pid_cpu_ticks "$BASHPID"
      printf '%s\n' "$reply" >"$worker_stage/CPU_TICKS.part"
      mv -- "$worker_stage/CPU_TICKS.part" "$worker_stage/CPU_TICKS"
      printf '%s\n' "$rc" >"$worker_stage/exit.part"
      mv -- "$worker_stage/exit.part" "$worker_stage/READY"
      exit "$rc"
    } >/dev/null 2>&1
    worker_pid=${V7_WORKER_PID-}
    case ${V7_WORKER[0]-} in '') ;; *) exec {V7_WORKER[0]}<&- ;; esac
    case ${V7_WORKER[1]-} in '') ;; *) exec {V7_WORKER[1]}>&- ;; esac
    if [[ $worker_pid =~ ^[1-9][0-9]*$ ]]; then
      pid_start_local "$worker_pid"
    else
      reply=
    fi
    if [[ $reply =~ ^[0-9]+$ ]]; then
      worker_start=$reply
    else
      case $worker_pid in
        [1-9][0-9]*) wait "$worker_pid" 2>/dev/null ;;
      esac
      worker_pid=
      return 125
    fi
    owned_add "$worker_pid" "$worker_start" || {
      stop_owned_local "$worker_pid" "$worker_start"
      worker_pid=
      return 125
    }
    worker_deadline=$((SECONDS + diag_to * (worker_count * 23 + worker_include_context * 12 + 2)))
  }
  finish_worker_budget() {
    local end_ms=$1 worker_ticks=${2-0} controller_now controller_delta=0
    local source_now source_delta=0 ticks elapsed
    [[ $worker_ticks =~ ^[0-9]+$ ]] || worker_ticks=0
    cpu_ticks_self
    controller_now=$reply
    if [[ $controller_now =~ ^[0-9]+$ ]] &&
         ((controller_now >= worker_ctrl_start)); then
      controller_delta=$((controller_now - worker_ctrl_start))
    fi
    ((worker_ticks > controller_delta)) && controller_delta=$worker_ticks
    if [[ $worker_source_cpu_tuple == "$source_pid:$source_start" ]]; then
      pid_start_local "$source_pid"
      if [[ $reply == "$source_start" ]]; then
        pid_cpu_ticks "$source_pid"
        source_now=$reply
        if [[ $source_now =~ ^[0-9]+$ ]] &&
             ((source_now >= worker_source_cpu_start)); then
          source_delta=$((source_now - worker_source_cpu_start))
        fi
      fi
    fi
    ticks=$((controller_delta + source_delta))
    reply=$(((ticks * 1000 + local_clk - 1) / local_clk))
    elapsed=$((end_ms - worker_started_ms))
    ((elapsed < 0)) && elapsed=0
    ((reply < elapsed)) && reply=$elapsed
    finish_observer_interval "$worker_started_ms" "$end_ms" "$reply"
  }
  consume_manifest() {
    local manifest=$1 prefix=$2 domain=$3 owner=$4 charge=$5
    local ci comp rc quality raw chain sid
    REFS=
    MANIFEST_COMPLETE=1
    while IFS=$'\t' read -r ci comp rc quality; do
      raw=$worker_stage/${prefix}_${ci}.raw
      [[ -f $raw ]] || { MANIFEST_COMPLETE=0; continue; }
      join4 "$domain" "$owner" "$comp" "v1_$quality"
      chain=$reply
      commit_state "$domain" "$chain" "$raw" \
        "transport_rc=$rc quality=$quality" 1 "$charge" 1 || return
      sid=$COMMIT_ID
      REFS+="${REFS:+ }$comp=$sid"
      case $quality in incomplete) MANIFEST_COMPLETE=0 ;; esac
    done <"$manifest"
  }
  refresh_active() {
    local manifest=$1 ci comp rc quality raw value key screen=unknown pair puser ppkg
    local -A fresh=()
    local -a fresh_keys=()
    while IFS=$'\t' read -r ci comp rc quality; do
      raw=$worker_stage/c_${ci}.raw
      [[ -f $raw ]] || continue
      case $comp in
        POWER|DISPLAY)
          if grep -Eiq 'Wakefulness=(Asleep|Dozing)|mWakefulness=(Asleep|Dozing)|mScreenOn=false|state=OFF|mScreenState=OFF' "$raw"; then
            screen=off
          else
            case $screen in
              off) ;;
              *)
                if grep -Eiq 'Wakefulness=Awake|mWakefulness=Awake|mScreenOn=true|state=ON|mScreenState=ON' "$raw"; then
                  screen=on
                fi
                ;;
            esac
          fi
          ;;
      esac
    done <"$manifest"
    while IFS=$'\t' read -r ci comp rc quality; do
      raw=$worker_stage/c_${ci}.raw
      [[ -f $raw ]] || continue
      case $comp in
        TELECOM|AUDIO) ;;
        ACTIVITIES|WINDOW) [[ $screen == on ]] || continue ;;
        *) continue ;;
      esac
      while IFS= read -r value; do
        case $value in
          pkg:*)
            value=${value#pkg:}
            valid_pkg "$value" || continue
            for pair in "${PKG_PAIRS[@]}"; do
              IFS=$'\x1f' read -r puser ppkg <<<"$pair"
              [[ $ppkg == "$value" ]] || continue
              key=upkg:$pair
              case ${fresh[$key]+set} in
                set) ;;
                *) fresh_keys+=("$key") ;;
              esac
              fresh[$key]=1
            done
            ;;
          pid:*)
            value=${value#pid:}
            [[ $value =~ ^[1-9][0-9]*$ ]] || continue
            key=pid:$value
            case ${fresh[$key]+set} in set) ;; *) fresh_keys+=("$key") ;; esac
            fresh[$key]=1
            ;;
          uid:*)
            value=${value#uid:}
            [[ $value =~ ^[0-9]+$ ]] || continue
            key=uid:$value
            case ${fresh[$key]+set} in set) ;; *) fresh_keys+=("$key") ;; esac
            fresh[$key]=1
            ;;
        esac
      done < <(awk '{
        s=$0
        while (match(s,/[A-Za-z0-9_]+([.][A-Za-z0-9_]+)+/)) {
          print "pkg:" substr(s,RSTART,RLENGTH)
          s=substr(s,RSTART+RLENGTH)
        }
        s=$0
        while (match(s,/(pid|PID|Pid)[=: ][[:space:]]*[0-9]+/)) {
          x=substr(s,RSTART,RLENGTH)
          sub(/[^0-9]*/,"",x)
          print "pid:" x
          s=substr(s,RSTART+RLENGTH)
        }
        s=$0
        while (match(s,/(uid|UID|Uid)[=: ][[:space:]]*[0-9]+/)) {
          x=substr(s,RSTART,RLENGTH)
          sub(/[^0-9]*/,"",x)
          print "uid:" x
          s=substr(s,RSTART+RLENGTH)
        }
      }' "$raw")
    done <"$manifest"
    ACTIVE=()
    for key in "${fresh_keys[@]}"; do ACTIVE[$key]=1; done
    active_at=$(date +%s)
  }
  reap_worker() {
    local i key tier quality start entity charge=0 body refs now_ms worker_ticks
    local finished_worker
    local snap done_epoch done_stamp done_day done_hour done_iso escaped_iso escaped_request
    [[ -n $worker_pid && -f $worker_stage/READY ]] || return 0
    finished_worker=$worker_pid
    wait "$finished_worker" 2>/dev/null
    owned_remove "$finished_worker"
    worker_pid=
    now_ms=$((SECONDS * 1000))
    if [[ -r $worker_stage/CPU_TICKS ]]; then
      worker_ticks=$(<"$worker_stage/CPU_TICKS")
    else
      worker_ticks=0
    fi
    finish_worker_budget "$now_ms" "$worker_ticks"
    snap=$(clock_snapshot)
    IFS='|' read -r done_epoch done_stamp done_day done_hour done_iso <<<"$snap"
    esc "$done_iso"; escaped_iso=$reply
    [[ $worker_class == O ]] && charge=1
    if ((worker_include_context)) && [[ -f $worker_stage/context.tsv ]]; then
      quality=$(<"$worker_stage/context.status")
      case $quality in complete) refresh_active "$worker_stage/context.tsv" ;; esac
      consume_manifest "$worker_stage/context.tsv" c context device "$charge" || return
      refs=$REFS
      body="epoch=$profile_epoch completed=$escaped_iso mode=$worker_context_mode quality=$quality states={$refs}"
      text_record CONTEXT "$body" "$charge" || return
      case $worker_context_mode:$quality in
        mandatory:complete)
          context_due=0
          context_attempt=0
          context_epoch_done=$profile_epoch
          ;;
        mandatory:*)
          if ((context_attempt == 0)); then
            context_attempt=1
            context_due=1
          else
            context_due=0
          fi
          ;;
        optional:*) ;;
      esac
      if waiting_optional_exists; then
        resolve_o_candidates "$quality"
      else
        o_refresh_due=0
      fi
    fi
    for ((i=1; i<=worker_count; i++)); do
      key=${B_JOB[$i]-}
      [[ -n $key ]] || continue
      read -r quality start <"$worker_stage/target_$i.status"
      entity=${JOB_ENTITY[$key]}
      tier=${JOB_TIER[$key]}
      refs=
      if [[ -f $worker_stage/target_$i.tsv ]]; then
        consume_manifest "$worker_stage/target_$i.tsv" "t_${i}" probe "$entity" \
          "$charge" || return
        refs=$REFS
        ((MANIFEST_COMPLETE)) || quality=incomplete
      fi
      esc "$entity"
      entity=$reply
      esc "${JOB_SNAPSHOT[$key]-}"; escaped_request=$reply
      body="epoch=${JOB_EPOCH[$key]} completed=$escaped_iso entity=$entity tier=$tier quality=$quality pid=${B_PID[$i]} uid=${B_UID[$i]} start=$start requested={$escaped_request} states={$refs}"
      text_record PROBE "$body" "$charge" || return
      case $tier:$quality in
        O:*) job_unset "$key" ;;
        *:terminal) job_unset "$key" ;;
        *:incomplete)
          if (( ${JOB_ATTEMPT[$key]-0} == 0 )); then
            JOB_ATTEMPT[$key]=1
            JOB_PHASE[$key]=retry_wait
            JOB_READY[$key]=$(( $(date +%s) + 2 ))
          else
            JOB_PHASE[$key]=exhausted
          fi
          ;;
        *) JOB_PHASE[$key]=done ;;
      esac
      unset 'B_JOB[$i]' 'B_PID[$i]' 'B_DEPTH[$i]' 'B_USER[$i]' \
        'B_PKG[$i]' 'B_UID[$i]' 'B_CMD[$i]'
    done
    case $worker_stage in "$runtime"/worker_*) rm -rf -- "$worker_stage" ;; esac
    worker_stage=
    worker_start=
    worker_child_note=
    worker_ctrl_start=0
    worker_source_cpu_start=0
    worker_source_cpu_tuple=
    worker_count=0
    worker_include_context=0
    worker_context_mode=none
  }
  cancel_worker() {
    local i key tier canceled_worker=$worker_pid canceled_start=$worker_start
    [[ -n $worker_pid ]] || return 0
    if owned_active "$canceled_worker"; then
      kill "$canceled_worker" 2>/dev/null
    fi
    stop_worker_child
    stop_owned_local "$canceled_worker" "$canceled_start"
    stop_worker_child
    owned_remove "$canceled_worker"
    finish_worker_budget "$((SECONDS * 1000))" 0
    for ((i=1; i<=worker_count; i++)); do
      key=${B_JOB[$i]-}
      [[ -n $key ]] || continue
      tier=${JOB_TIER[$key]-O}
      case $tier in
        O) job_unset "$key" ;;
        *)
          JOB_PHASE[$key]=queued
          append_retry "$key"
          ;;
      esac
      unset 'B_JOB[$i]' 'B_PID[$i]' 'B_DEPTH[$i]' 'B_USER[$i]' \
        'B_PKG[$i]' 'B_UID[$i]' 'B_CMD[$i]'
    done
    case $worker_context_mode in
      optional) resolve_o_candidates canceled ;;
    esac
    case $worker_stage in "$runtime"/worker_*) rm -rf -- "$worker_stage" ;; esac
    worker_pid=
    worker_start=
    worker_stage=
    worker_child_note=
    worker_ctrl_start=0
    worker_source_cpu_start=0
    worker_source_cpu_tuple=
    worker_count=0
    worker_include_context=0
    worker_context_mode=none
  }
  drain_mandatory() {
    local why=$1 deadline=$((SECONDS + diag_to * 2 + 5))
    case $worker_class:$worker_context_mode in
      O:*|M:mandatory) cancel_worker ;;
    esac
    while mandatory_exists; do
      if [[ -n $worker_pid ]]; then
        if [[ -f $worker_stage/READY ]]; then
          reap_worker || return
        elif ((SECONDS >= deadline)); then
          cancel_worker
          break
        else
          sleep 0.1
        fi
      else
        launch_worker || return
        [[ -n $worker_pid ]] || break
      fi
    done
    if [[ -n $worker_pid && -f $worker_stage/READY ]]; then
      reap_worker || return
    fi
  }

  profile_job_epoch() {
    case $refresh in
      hourly) reply=$out_hour ;;
      daily) reply=$out_day ;;
      off) reply=$invocation_epoch ;;
    esac
  }
  request_optional() {
    local entity=$1 snapshot=$2 key
    ((watch && context)) || return 0
    observer_open || return 0
    mandatory_exists && return 0
    profile_job_epoch
    job_key "$reply" "$entity" O
    key=$reply
    [[ -z ${JOB_PHASE[$key]-} ]] || return 0
    profile_job_epoch
    enqueue_job "$reply" "$entity" O "$snapshot"
    key=$reply
    if active_fresh; then
      if entity_active "$entity" "${JOB_PID[$key]-0}" "${JOB_UID[$key]--}" \
           "${JOB_USER[$key]--}" "${JOB_PKG[$key]--}"; then
        job_unset "$key"
      fi
    else
      JOB_PHASE[$key]=context_wait
      o_refresh_due=1
    fi
  }
  normalize_thresholds() {
    local max fallback_low=1000 fallback_high=9000
    ((topmax100 > 0)) || topmax100=$((device_nproc * 10000))
    max=$topmax100
    low100=$low_arg
    high100=$high_arg
    if ((low100 < 0 || high100 < low100 || high100 > max)); then
      low100=$fallback_low
      high100=$fallback_high
      if ((topmax_genuine)); then
        ((high100 > max)) && high100=$max
        ((low100 > high100)) && low100=$high100
      fi
    fi
    thresholds_ready=1
  }
  save_selected_top() {
    local stage=$1 observation=$2 partial=$3 raw_only=$4 truncated=$5
    local chain tuple sid
    join4 top device raw v7
    chain=$reply
    commit_state top "$chain" "$stage" "$observation" 0 0 0 || return
    sid=$COMMIT_ID
    if ((COMMIT_NEW)); then
      init_occurrence "$sid"
      saved_n=$((saved_n + 1))
    else
      occurrence_tuple "$frame_iso" "$source_index" "$partial" "$raw_only" \
        "$truncated" "$frame_drop_bytes" "$frame_drop_lines"
      tuple=$reply
      add_top_occurrence "$sid" "$tuple" || return
    fi
    selected_n=$((selected_n + 1))
    last_selected_epoch=$frame_epoch
  }
  finish_frame() {
    local partial=${1-0} line pid uid_col user pr ni virt res shr state cpu mem timep args
    local saw=0 rows_seen=0 bad=0 ambiguous=0 cpu100 key hot=0 select=0 raw_only=0
    local observation snapshot
    local -a order=()
    local -A sum=() hottest=() hotpid=() hotkind=() hotlabel=()
    local -A hotuser=() hotuid=() hotpkg=() hotcmd=()
    [[ -n $frame_file && -f $frame_file ]] || return 0
    snapshot_capture_helpers
    while IFS= read -r line; do
      line=${line%$'\r'}
      if [[ $line =~ ^[[:space:]]*([0-9]+)%cpu ]]; then
        topmax100=$((10#${BASH_REMATCH[1]} * 100))
        topmax_genuine=1
        thresholds_ready=0
      fi
      if [[ $line =~ ^[[:space:]]*PID[[:space:]]+UID[[:space:]]+USER[[:space:]]+PR[[:space:]]+NI[[:space:]]+VIRT[[:space:]]+RES[[:space:]]+SHR[[:space:]]+S[[:space:]]+%CPU[[:space:]]+%MEM[[:space:]]+TIME\+[[:space:]]+ARGS[[:space:]]*$ ]]; then
        saw=1
        continue
      fi
      ((saw)) || continue
      [[ -n $line ]] || continue
      read -r pid uid_col user pr ni virt res shr state cpu mem timep args <<<"$line"
      if [[ $pid =~ ^[1-9][0-9]*$ && $uid_col =~ ^[0-9]+$ ]] &&
           parse_pct100 "$cpu" && [[ -n $args ]]; then
        cpu100=$reply
        ((rows_seen++))
        profiler_pid "$pid" && continue
        resolve_entity "$user" "$args" "$pid" "$uid_col"
        key=$TARGET_KEY
        case $TARGET_KIND in ambiguous) ((ambiguous++)) ;; esac
        if [[ -z ${sum[$key]-} ]]; then
          order+=("$key")
          sum[$key]=0
          hottest[$key]=-1
        fi
        sum[$key]=$((sum[$key] + cpu100))
        if ((cpu100 > hottest[$key])); then
          hottest[$key]=$cpu100
          hotpid[$key]=$pid
          hotkind[$key]=$TARGET_KIND
          hotlabel[$key]=$TARGET_LABEL
          hotuser[$key]=$TARGET_USER
          hotuid[$key]=$TARGET_UID
          hotpkg[$key]=$TARGET_PKG
          hotcmd[$key]=$TARGET_CMD
        fi
      else
        ((bad++))
      fi
    done <"$frame_file"
    if ((partial == 0 && saw && rows_seen)); then
      ((source_healthy_frames++))
      if ((source_healthy_frames >= 3)); then source_retry=0; fi
    else
      source_healthy_frames=0
    fi
    ((thresholds_ready)) || normalize_thresholds
    ((saw && rows_seen)) || raw_only=1
    for key in "${order[@]}"; do
      remember_entity "$key" "${hotkind[$key]}" "${hotlabel[$key]}" \
        "${hotuser[$key]}" "${hotuid[$key]}" "${hotpkg[$key]}" \
        "${hotcmd[$key]}" "${hotpid[$key]}"
      ((sum[$key] >= low100)) && hot=1
    done
    if ((partial || frame_drop_bytes || raw_only || hot)); then
      select=1
    elif (((source_index - 1) % every == 0)); then
      if ((period == 0 || last_selected_epoch == 0 ||
            frame_epoch - last_selected_epoch >= period)); then
        select=1
      fi
    fi
    observation="captured=$frame_iso source=$source_index partial=$partial raw_only=$raw_only truncated=$((frame_drop_bytes > 0)) bad_rows=$bad ambiguous_rows=$ambiguous drop_bytes=$frame_drop_bytes drop_lines=$frame_drop_lines"
    if ((select)); then
      save_selected_top "$frame_file" "$observation" "$partial" "$raw_only" \
        "$((frame_drop_bytes > 0))" || return
    else
      rm -f -- "$frame_file"
    fi
    for key in "${order[@]}"; do
      snapshot="source=$source_index pid=${hotpid[$key]} uid=${ENTITY_UID[$key]} cpu100=${sum[$key]}"
      profile_job_epoch
      if ((sum[$key] >= high100)); then
        enqueue_job "$reply" "$key" H "$snapshot"
      elif ((sum[$key] >= low100)); then
        enqueue_job "$reply" "$key" L "$snapshot"
      fi
    done
    for key in "${order[@]}"; do
      ((sum[$key] >= low100)) || continue
      request_optional "$key" \
        "source=$source_index pid=${hotpid[$key]} uid=${ENTITY_UID[$key]} cpu100=${sum[$key]}"
    done
    frame_file=
    frame_open=0
    if ((max_reports > 0 && selected_n >= max_reports)); then
      set_reason max_reports
    fi
  }
  append_frame_line() {
    local line=$1 n=$(( ${#1} + 1 ))
    frame_total=$((frame_total + n))
    if ((frame_kept + n <= frame_cap)); then
      printf '%s\n' "$line" >>"$frame_file"
      frame_kept=$((frame_kept + n))
    else
      frame_drop_bytes=$((frame_drop_bytes + n))
      frame_drop_lines=$((frame_drop_lines + 1))
    fi
  }
  append_frame_fragment() {
    local fragment=$1 n=${#1}
    frame_total=$((frame_total + n))
    if ((frame_kept + n <= frame_cap)); then
      printf '%s' "$fragment" >>"$frame_file"
      frame_kept=$((frame_kept + n))
    else
      frame_drop_bytes=$((frame_drop_bytes + n))
    fi
  }
  checkpoint_now() {
    local why=$1
    flush_occurrences || return
    text_record CHECKPOINT \
      "reason=$why source=$source_index selected=$selected_n saved=$saved_n records=$record_seq file_bytes=$file_bytes run_bytes=$run_bytes duty_debt_ms=$duty_debt" 0 ||
      return
    next_checkpoint=$((SECONDS + checkpoint_interval))
  }
  finish_day() {
    local why=$1
    ((output_open)) || return 0
    case $why in
      day_rotate|duration|until|max_reports|signal_*) drain_mandatory "$why" || return ;;
    esac
    if [[ -n $worker_pid ]]; then
      if [[ -f $worker_stage/READY ]]; then reap_worker; else cancel_worker; fi
    fi
    flush_occurrences || return
    cache_end "$why" 0 || return
    disclose_jobs "$why" || return
    text_record DAY_END \
      "day=$out_day reason=$why records=$record_seq selected=$selected_n file_bytes=$file_bytes" 0 ||
      return
    exec {ofd}>&-
    ofd=
    output_open=0
  }
  clear_identity_state() {
    ENTITY_KIND=()
    ENTITY_LABEL=()
    ENTITY_USER=()
    ENTITY_UID=()
    ENTITY_PKG=()
    ENTITY_CMD=()
    ENTITY_PID=()
    ENTITY_KEYS=()
    ACTIVE=()
    active_at=0
    thresholds_ready=0
    topmax100=0
    topmax_genuine=0
    remote_top_pid=
    remote_top_start=
  }
  handle_boot_change() {
    local new_boot=$1 old_boot=$boot_id body=$runtime/boot_change_$RANDOM
    if [[ -n $worker_pid ]]; then
      if [[ -f $worker_stage/READY ]]; then reap_worker || return; else cancel_worker; fi
    fi
    cache_end boot_change 1 || return
    disclose_jobs boot_change || return
    printf 'old_boot=%s\tnew_boot=%s\tidentity_cache_jobs_inventory=reset\n' \
      "$old_boot" "$new_boot" >"$body"
    write_record BOOT_CHANGE "$body" 0 || {
      local rc=$?
      rm -f -- "$body"
      return "$rc"
    }
    rm -f -- "$body"
    reset_jobs
    clear_identity_state
    boot_id=$new_boot
    adb_uid=$RECHECK_UID
    device_nproc=$RECHECK_NPROC
    remote_clk=$RECHECK_CLK
    capture_inventory
    write_record INVENTORY "$inventory_file" 0 || return
    context_due=$context
    context_attempt=0
    context_epoch_done=
    o_refresh_due=0
  }
  revalidate_device_identity() {
    read_device_identity || {
      set_reason reconnect_identity_error
      return 2
    }
    [[ $RECHECK_UID == "$adb_uid" ]] || {
      set_reason reconnect_uid_changed
      return 2
    }
    if [[ $RECHECK_BOOT == "$boot_id" ]]; then
      device_nproc=$RECHECK_NPROC
      remote_clk=$RECHECK_CLK
    else
      handle_boot_change "$RECHECK_BOOT" || return
    fi
  }
  scope_transition() {
    local epoch=$1 stamp=$2 day=$3 hour=$4 iso=$5
    if ((output_open == 0)); then
      open_day "$day" "$hour" "$stamp"
      return
    fi
    if [[ $day == "$out_day" ]]; then
      :
    else
      finish_day day_rotate || return
      case $refresh in
        off) ;;
        *) reset_jobs ;;
      esac
      hour_bytes=0
      o_hour_bytes=0
      open_day "$day" "$hour" "$stamp" || return
      expire_terminal_jobs
      return
    fi
    if [[ $hour == "$out_hour" ]]; then
      :
    else
      cache_end hour_rotate || return
      out_hour=$hour
      hour_bytes=0
      o_hour_bytes=0
      case $refresh in hourly) profile_epoch=$hour ;; esac
      expire_terminal_jobs
      context_due=$context
      context_attempt=0
      context_epoch_done=
    fi
  }
  check_limits() {
    local now
    [[ $reason == running ]] || return
    if ((duration > 0 && SECONDS - run_start_seconds >= duration)); then
      set_reason duration
      return
    fi
    if ((until > 0)); then
      now=$(date +%s)
      if ((now >= until)); then
        set_reason until
        return
      fi
    fi
    if ((max_bytes > 0 && run_bytes >= max_bytes)); then
      hard_cap_closed=1
      set_reason max_bytes
    fi
  }
  clip_wait() {
    local wanted=$1 remain now
    reply=$wanted
    if ((duration > 0)); then
      remain=$((duration - (SECONDS - run_start_seconds)))
      ((remain < reply)) && reply=$remain
    fi
    if ((until > 0)); then
      now=$(date +%s)
      remain=$((until - now))
      ((remain < reply)) && reply=$remain
    fi
    ((reply < 0)) && reply=0
  }

  read -r -d '' REMOTE_TOP <<'V7_REMOTE_TOP'

case $delay in

  top_column_preflight() {
    local probe=$runtime/top_preflight.raw
    ro "$diag_to" top -b -n 1 -m 1 -o "$TOP_FIELDS" -s 10 \
      "${TOP_EXTRA[@]}" >"$probe" 2>&1 || return 1
    awk '
      $1=="PID" && $2=="UID" && $3=="USER" && $4=="PR" &&
      $5=="NI" && $6=="VIRT" && $7=="RES" && $8=="SHR" &&
      $9=="S" && $10=="%CPU" && $11=="%MEM" &&
      $12=="TIME+" && $13=="ARGS" && NF==13 {header=1; next}
      header && $1 ~ /^[1-9][0-9]*$/ && $2 ~ /^[0-9]+$/ &&
        $10 ~ /^[0-9]+([.][0-9]+)?$/ && NF>=13 {row=1}
      END {exit (header && row) ? 0 : 1}
    ' "$probe"
  }
  start_source() {
    local script=$runtime/top_source.sh note=$runtime/source_owner_$RANDOM p s
    ((++source_attempt))
    revalidate_device_identity || return
    top_column_preflight || {
      set_reason top_layout_error
      return 2
    }
    printf '%s\n' "$REMOTE_TOP" >"$script" || return 1
    exec {source_fd}< <(
      pid_start_local "$BASHPID" || exit 125
      printf '%s %s\n' "$BASHPID" "$reply" >"$note.part" || exit 125
      mv -- "$note.part" "$note" || exit 125
      exec "${A[@]}" shell -T sh -s "$shell_uid" "$rows" "$delay" \
        "$TOP_FIELDS" "${TOP_EXTRA[@]}" <"$script" 2>&1
    ) || return 1
    wait_pid_note "$note"
    if (($?)); then
      if [[ $source_fd =~ ^[0-9]+$ ]]; then
        exec {source_fd}<&-
        source_fd=
      fi
      if [[ -r $note ]]; then
        read -r p s <"$note"
        if [[ $p =~ ^[1-9][0-9]*$ && $s =~ ^[0-9]+$ ]]; then
          stop_owned_local "$p" "$s"
        fi
      fi
      rm -f -- "$note" "$note.part"
      return 125
    fi
    source_pid=$reply
    source_start=$reply2
    owned_add "$source_pid" "$source_start" || {
      stop_owned_local "$source_pid" "$source_start"
      if [[ $source_fd =~ ^[0-9]+$ ]]; then
        exec {source_fd}<&-
        source_fd=
      fi
      source_pid=
      source_start=
      rm -f -- "$note" "$note.part"
      return 125
    }
    remote_top_pid=
    remote_top_start=
    source_owner_seen=0
    source_healthy_frames=0
    source_noise=
    return 0
  }
  read -r -d '' REMOTE_STOP_TOP <<'V7_REMOTE_STOP_TOP'
guard=$1
pid=$2
expected=$3
actual=$(id -u 2>/dev/null)
if printf '%s\n' "$actual" | grep -Eq '^[0-9]+$'; then :; else exit 46; fi
case $guard in any) ;; *) [ "$actual" = "$guard" ] || exit 46 ;; esac
same_tuple() {
  [ -r "/proc/$pid/stat" ] || return 1
  s=$(cat "/proc/$pid/stat" 2>/dev/null) || return 1
  tail=${s##*) }
  set -- $tail
  [ "$#" -ge 20 ] || return 1
  [ "${20}" = "$expected" ]
}

  sleep 0.1
  i=$((i + 1))
done

kill -KILL "$pid" 2>/dev/null

  stop_remote_top() {
    [[ $remote_top_pid =~ ^[1-9][0-9]*$ ]] || return 0
    [[ $remote_top_start =~ ^[0-9]+$ ]] || return 0
    timeout -k 2s "$((diag_to + 5))s" "${A[@]}" shell -T sh -s \
      "$shell_uid" "$remote_top_pid" "$remote_top_start" \
      <<<"$REMOTE_STOP_TOP" >/dev/null 2>&1
    remote_top_pid=
    remote_top_start=
  }
  stop_source() {
    local rc=127
    stop_remote_top
    if [[ $source_fd =~ ^[0-9]+$ ]]; then
      exec {source_fd}<&-
      source_fd=
    fi
    if [[ $source_pid =~ ^[1-9][0-9]*$ ]]; then
      stop_owned_local "$source_pid" "$source_start"
      rc=$?
      owned_remove "$source_pid"
    fi
    source_pid=
    source_start=
    source_owner_seen=0
    SOURCE_RC=$rc
  }
  source_event() {
    local read_rc=$1 source_rc=$2 snap epoch stamp day hour iso noise=
    snap=$(clock_snapshot)
    IFS='|' read -r epoch stamp day hour iso <<<"$snap"
    scope_transition "$epoch" "$stamp" "$day" "$hour" "$iso" || return
    if [[ -n $source_noise ]]; then esc "$source_noise"; noise=" preframe=$reply"; fi
    text_record SOURCE_EVENT \
      "attempt=$source_attempt read_rc=$read_rc source_rc=$source_rc retry=$source_retry$noise" 0
    source_noise=
  }
  on_signal() {
    set_reason signal_$1
    if [[ $source_pid =~ ^[1-9][0-9]*$ ]]; then
      pid_start_local "$source_pid"
      [[ $reply == "$source_start" ]] && kill "$source_pid" 2>/dev/null
    fi
  }
  finish_once() {
    local status=${1-0} snap epoch stamp day hour iso clean_finish=1
    ((finish_done == 0)) || return "$status"
    finish_done=1
    trap - HUP INT TERM
    if ((frame_open)); then finish_frame 1; fi
    stop_source
    snap=$(clock_snapshot)
    IFS='|' read -r epoch stamp day hour iso <<<"$snap"
    case $reason in max_bytes|write_error) clean_finish=0 ;; esac
    if ((output_open && clean_finish)); then
      scope_transition "$epoch" "$stamp" "$day" "$hour" "$iso"
      checkpoint_now final
      finish_day "$reason"
    elif ((output_open)); then
      if [[ -n $worker_pid ]]; then cancel_worker; fi
      exec {ofd}>&-
      ofd=
      output_open=0
    elif [[ -n $worker_pid ]]; then
      cancel_worker
    fi
    clear_prev
    case $runtime in
      "${TMPDIR:-/tmp}"/phone_top_v7.*) rm -rf -- "$runtime" ;;
    esac
    printf 'Stopped: %s; source=%s selected=%s saved=%s bytes=%s%s\n' \
      "$reason" "$source_index" "$selected_n" "$saved_n" "$run_bytes" \
      "${file:+; last=$file}"
    case $reason in
      signal_1) status=129 ;;
      signal_2) status=130 ;;
      signal_15) status=143 ;;
      *_error|write_error|create_error) status=1 ;;
    esac
    return "$status"
  }

  trap 'on_signal 1' HUP
  trap 'on_signal 2' INT
  trap 'on_signal 15' TERM

  connect_device || {
    set_reason connect_error
    finish_once 2
    return $?
  }
  top_column_preflight || {
    set_reason top_layout_error
    finish_once 2
    return $?
  }
  snap=$(clock_snapshot)
  IFS='|' read -r now_epoch now_stamp now_day now_hour now_iso <<<"$snap"
  if ((until > 0 && until <= now_epoch)); then
    set_reason until
  else
    scope_transition "$now_epoch" "$now_stamp" "$now_day" "$now_hour" "$now_iso" ||
      set_reason open_error
  fi
  if [[ $reason == running ]]; then
    launch_worker || set_reason worker_launch_error
    if [[ $reason == running ]]; then
      start_source || set_reason source_start_error
    fi
  fi

  while [[ $reason == running ]]; do
    line=
    clip_wait "$effective_source_to"
    read_timeout=$reply
    if ((read_timeout == 0)); then
      check_limits
      break
    fi
    if IFS= read -r -t "$read_timeout" -u "$source_fd" line; then
      if ((frame_open == 0 && source_owner_seen == 0)) &&
           [[ $line =~ ^V7_TOP_OWNER[[:space:]]+([1-9][0-9]*)[[:space:]]+([0-9]+)$ ]]; then
        remote_top_pid=${BASH_REMATCH[1]}
        remote_top_start=${BASH_REMATCH[2]}
        source_owner_seen=1
      elif [[ $line == Tasks:* || $line == Threads:* ]]; then
        if ((frame_open)); then finish_frame 0 || set_reason frame_error; fi
        [[ $reason == running ]] || break
        if [[ -n $worker_pid && -f $worker_stage/READY ]]; then
          reap_worker || set_reason worker_reap_error
        elif [[ -n $worker_pid ]] && ((SECONDS >= worker_deadline)); then
          cancel_worker
        fi
        [[ $reason == running ]] || break
        snap=$(clock_snapshot)
        IFS='|' read -r now_epoch now_stamp now_day now_hour now_iso <<<"$snap"
        scope_transition "$now_epoch" "$now_stamp" "$now_day" "$now_hour" "$now_iso" ||
          set_reason scope_error
        [[ $reason == running ]] || break
        if ((SECONDS >= next_checkpoint)); then
          checkpoint_now periodic || set_reason checkpoint_error
        fi
        check_limits
        [[ $reason == running ]] || break
        launch_worker || set_reason worker_launch_error
        [[ $reason == running ]] || break
        ((source_index++))
        frame_epoch=$now_epoch
        frame_stamp=$now_stamp
        frame_day=$now_day
        frame_hour=$now_hour
        frame_iso=$now_iso
        frame_file=$runtime/frame_$source_index.raw
        : >"$frame_file" || { set_reason frame_stage_error; break; }
        frame_open=1
        frame_kept=0
        frame_total=0
        frame_drop_bytes=0
        frame_drop_lines=0
        append_frame_line "$line"
      elif ((frame_open)); then
        append_frame_line "$line"
      elif ((${#source_noise} < 4096)); then
        source_noise+="${source_noise:+$'\n'}$line"
      fi
    else
      read_rc=$?
      if ((frame_open)); then
        case $line in '') ;; *) append_frame_fragment "$line" ;; esac
        finish_frame 1 || set_reason frame_error
      elif [[ -n $line && ${#source_noise} -lt 4096 ]]; then
        source_noise+="${source_noise:+$'\n'}$line"
      fi
      stop_source
      check_limits
      if [[ $reason == running ]]; then
        source_event "$read_rc" "$SOURCE_RC" || set_reason source_event_error
      fi
      if [[ -n $worker_pid && -f $worker_stage/READY ]]; then
        reap_worker || set_reason worker_reap_error
      fi
      check_limits
      [[ $reason == running ]] || break
      case $source_retry in
        0) backoff=1 ;;
        1) backoff=2 ;;
        2) backoff=4 ;;
        3) backoff=8 ;;
        4) backoff=16 ;;
        *) backoff=30 ;;
      esac
      ((source_retry++))
      clip_wait "$backoff"
      backoff=$reply
      if ((backoff > 0)); then sleep "$backoff"; fi
      check_limits
      [[ $reason == running ]] || break
      start_source || set_reason source_restart_error
    fi
    check_limits
  done

  status=0
  finish_once "$status"

)
