using System.Threading;

    static readonly object gate = new object();
    static readonly ManualResetEvent ready = new ManualResetEvent(false);

    static RemapWindow window;

    public static string Start()
    {
        lock (gate) {
            if (worker != null && worker.IsAlive)
                return "Already active. Ctrl+Alt+Q stops it.";

            DisableQuickEdit();

            ready.Reset();
            startupStatus = null;

            worker = new Thread(ThreadMain);
            worker.IsBackground = true;
            worker.SetApartmentState(ApartmentState.STA);
            worker.Start();
        }

        if (!ready.WaitOne(5000)) {
            Stop();
            return "Startup timed out.";
        }

        return startupStatus;
    }

    public static string Stop()
    {
        Thread activeWorker;
        RemapWindow activeWindow;

        lock (gate) {
            activeWorker = worker;
            activeWindow = window;
        }

        if (activeWorker == null || !activeWorker.IsAlive) {
            RestoreQuickEdit();
            return "Already stopped.";
        }

        try {
            if (activeWindow != null && activeWindow.IsHandleCreated) {
                activeWindow.BeginInvoke(
                    (MethodInvoker)delegate { activeWindow.Close(); }
                );
            }
        }
        catch { }

        if (Thread.CurrentThread != activeWorker)
            activeWorker.Join(3000);

        return activeWorker.IsAlive ? "Stop request sent." : "Stopped.";
    }

        get {
            lock (gate) {
                return worker != null &&
                       worker.IsAlive &&
                       window != null &&
                       window.HookActive;
            }
        }

    {

            RemapWindow activeWindow = new RemapWindow();

            lock (gate)

            Application.Run(activeWindow);

        catch (Exception error) {

            ready.Set();

                window = null;
                worker = null;

        const uint RIGHTUP   = 0x0010;

            ShowInTaskbar = false;
            Opacity = 0;
            WindowState = FormWindowState.Minimized;

            hook = SetWindowsHookEx(
                WH_MOUSE_LL,
                mouseProc,
                GetModuleHandle(null),
                0
            );

            int error = hook == IntPtr.Zero

                : 0;

                message.WParam.ToInt32() == 1) {
                Close();
                return;

            base.WndProc(ref message);

        IntPtr MouseHook(int code, IntPtr wParam, IntPtr lParam)
        {

                int message = wParam.ToInt32();

                if (message == WM_MBUTTONDOWN) {
                    if (!rightHeld) {
                        EmitRight(RIGHTDOWN);
                        rightHeld = true;

                    return new IntPtr(1);

                if (message == WM_MBUTTONUP) {
                    if (rightHeld) {
                        EmitRight(RIGHTUP);
                        rightHeld = false;
                    }

                    return new IntPtr(1);
                }

            return CallNextHookEx(hook, code, wParam, lParam);

                rightHeld = false;

                UnregisterHotKey(Handle, 1);
